1"""The ``secrets`` command group, as a factory other applications can embed. 

2 

3:func:`build_secrets_app` returns a fresh :class:`typer.Typer` each call, so a 

4host CLI can mount these commands under its own name, render them through its 

5own console, and — the point of the exercise — supply the SSM search path its 

6users should not have to type: 

7 

8```python 

9import typer 

10 

11from pdum.aws.cli import build_secrets_app 

12 

13app = typer.Typer() 

14app.add_typer(build_secrets_app(default_path="/acme/:/org/", envvar="ACME_SSM_PATH"), name="secrets") 

15``` 

16 

17``acme secrets list`` now works bare. The search path resolves in this order: 

18 

191. ``--path`` on the command line; 

202. the environment variable named by *envvar*; 

213. *default_path*, which may be a callable when the host reads it from a 

22 config file and wants that read deferred to invocation time; 

234. otherwise an error — the library itself ships no default, because a shared 

24 one would let unrelated projects collide in one namespace. 

25 

26A path is one or more SSM prefixes, colon-separated, most specific first — 

27reads fall back along it, writes and deletes target the first prefix only 

28(see :mod:`pdum.aws.secrets`). Pass ``expose_path_option=False`` to drop the 

29flag entirely and pin the namespace to *default_path*. 

30""" 

31

47#: Key under which the group callback stashes the store for its subcommands. 

48#: 

49#: ``ctx.meta`` rather than ``ctx.obj``: meta is one dict shared by the entire 

50#: Click context stack, so mounting this group inside a host application never 

51#: shadows whatever that application put in ``ctx.obj`` for its own commands. 

52STORE_KEY = "pdum.aws.secrets.store" 

53 

54#: Keys never pushed into the store by ``secrets import``. Putting AWS 

55#: credentials into a store you need AWS credentials to read would be circular. 

56SKIP_ON_IMPORT = frozenset( 

57 { 

58 "AWS_ACCESS_KEY_ID", 

59 "AWS_SECRET_ACCESS_KEY", 

60 "AWS_SESSION_TOKEN", 

61 "AWS_DEFAULT_REGION", 

62 "AWS_REGION", 

63 "AWS_PROFILE", 

64 } 

65) 

66 

67#: A search path supplied by the host: literal, deferred, or absent. 

68PathDefault = str | Callable[[], str | None] | None

71def store_from(ctx: typer.Context) -> SecretStore: 

72 """Return the store this invocation resolved. 

73 

74 Useful to host applications adding their own commands to a group built by 

75 :func:`build_secrets_app`, which then read the same search path as the rest. 

76 

77 Parameters 

78 ---------- 

79 ctx : typer.Context 

80 Context of a command inside the group. 

81 

82 Returns 

83 ------- 

84 SecretStore 

85 The store built by the group's callback. 

86 """ 

87 return ctx.meta[STORE_KEY]

100 return ...

104def _describe_table(rows: list[dict[str, Any]], *, show_origin: bool) -> Table: 

105 """Render ``SecretStore.describe`` output as a table. 

106 

107 Shadowed rows — reachable layers a more specific one hides — render dim, 

108 so the listing shows both what resolves and what it eclipses. 

109 """ 

110 table = Table(box=None, pad_edge=False) 

111 table.add_column("NAME", style="bold") 

112 table.add_column("TYPE") 

113 table.add_column("VER", justify="right") 

114 table.add_column("MODIFIED") 

115 if show_origin: 

116 table.add_column("ORIGIN") 

117 for row in rows: 

118 cells = [ 

119 row["name"], 

120 "secure" if row["type"] == "SecureString" else "plain", 

121 str(row["version"]), 

122 row["modified"].strftime("%Y-%m-%d %H:%M"), 

123 ] 

124 if show_origin: 

125 cells.append(row["origin"] + (" (shadowed)" if row["shadowed"] else "")) 

126 table.add_row(*cells, style="dim" if row["shadowed"] else None) 

127 return table

130def build_secrets_app( 

131 *, 

132 console: Console | None = None, 

133 default_path: PathDefault = None, 

134 envvar: str | None = "PDUM_SSM_PATH", 

135 expose_path_option: bool = True, 

136 store_factory: Callable[[str], SecretStore] = SecretStore, 

137 help: str = "Manage secrets in AWS SSM Parameter Store.", 

138) -> typer.Typer: 

139 """Build a ``secrets`` command group. 

140 

141 Parameters 

142 ---------- 

143 console : rich.console.Console, optional 

144 Console every command in the group renders through; defaults to 

145 :data:`~pdum.aws.cli.output.DEFAULT_CONSOLE`. 

146 default_path : str or callable, optional 

147 SSM search path to use when neither the flag nor the environment 

148 supplies one. Pass a zero-argument callable to defer resolving it — a 

149 host reading the path from a config file wants that read to happen on 

150 invocation, not on import. 

151 envvar : str, optional 

152 Environment variable consulted before *default_path*. Pass ``None`` 

153 to disable environment lookup entirely. 

154 expose_path_option : bool, default True 

155 Whether to offer ``--path``. ``False`` removes the flag, pinning the 

156 group to *default_path*. 

157 store_factory : callable, default :class:`~pdum.aws.secrets.SecretStore` 

158 Called with the resolved path to build the store. Override to thread 

159 host configuration through, e.g. 

160 ``lambda path: SecretStore(path, region=cfg.region)``. 

161 help : str 

162 Group help text. 

163 

164 Returns 

165 ------- 

166 typer.Typer 

167 A new application, ready to pass to ``add_typer``. 

168 """ 

169 out = console or DEFAULT_CONSOLE 

170 app = typer.Typer(help=help, no_args_is_help=True)

172 def configure(ctx: typer.Context, path: str | None) -> None: 

173 """Resolve the search path and put the store where subcommands find it.""" 

174 if not path: 

175 path = default_path() if callable(default_path) else default_path 

176 if not path: 

177 source = f" or set {envvar}" if envvar else "" 

178 out.print(f"[red]no SSM search path configured: pass --path{source}[/]") 

179 raise typer.Exit(2) 

180 try: 

181 ctx.meta[STORE_KEY] = store_factory(path) 

182 except ValueError as exc: 

183 out.print(f"[red]{exc}[/]") 

184 raise typer.Exit(2) from exc

209 @app.command("list") 

210 def secrets_list( 

211 ctx: typer.Context, 

212 long: bool = typer.Option(False, "--long", "-l", help="show type, version, last-modified and origin"), 

213 ) -> None: 

214 """List secret names visible on the search path.""" 

215 store = store_from(ctx) 

216 if not long: 

217 for name in sorted(store.names()): 

218 emit(out, name) 

219 return 

220 

221 rows = sorted(store.describe(), key=lambda r: (r["name"], r["shadowed"])) 

222 if not rows: 

223 out.print(f"[yellow]no secrets under {':'.join(store.prefixes)}[/]") 

224 return 

225 out.print(_describe_table(rows, show_origin=len(store.prefixes) > 1))

239 @app.command("set") 

240 def secrets_set( 

241 ctx: typer.Context, 

242 name: str = typer.Argument(...), 

243 value: str = typer.Argument(None, help="value; omit or pass '-' to read stdin"), 

244 plain: bool = typer.Option(False, "--plain", help="store as String rather than SecureString"), 

245 ) -> None: 

246 """Set a secret in the first prefix, taking the value from the argument or stdin. 

247 

248 Prefer stdin: a value passed as an argument lands in your shell history. 

249 """ 

250 store = store_from(ctx) 

251 if value in (None, "-"): 

252 value = sys.stdin.read().rstrip("\n") 

253 if not value: 

254 out.print("[red]refusing to store an empty value[/]") 

255 raise typer.Exit(1) 

256 store.put(name, value, secure=not plain) 

257 out.print(f"[green]set[/] {store.path(name)}")

259 @app.command("rm") 

260 def secrets_rm( 

261 ctx: typer.Context, 

262 name: str = typer.Argument(...), 

263 yes: bool = typer.Option(False, "--yes", "-y", help="skip the confirmation prompt"), 

264 ) -> None: 

265 """Delete a secret from the first prefix. 

266 

267 A secret living only in a fallback layer is refused, naming where it 

268 is: deleting from a shared layer takes an explicit ``--path`` there. 

269 """ 

270 store = store_from(ctx) 

271 if not yes: 

272 typer.confirm(f"delete {store.path(name)}?", abort=True) 

273 try: 

274 store.delete(name) 

275 except KeyError as exc: 

276 out.print(f"[red]{exc.args[0]}[/]") 

277 raise typer.Exit(1) from exc 

278 out.print(f"[green]deleted[/] {store.path(name)}")

280 @app.command("import") 

281 def secrets_import( 

282 ctx: typer.Context, 

283 path: Path = typer.Argument(Path(".secrets"), help="a KEY=VALUE file"), 

284 dry_run: bool = typer.Option(False, "--dry-run", help="show what would be pushed, write nothing"), 

285 ) -> None: 

286 """Push secrets from a KEY=VALUE file into the first prefix. 

287 

288 AWS bootstrap keys are skipped; see :data:`SKIP_ON_IMPORT`. 

289 """ 

290 store = store_from(ctx) 

291 if not path.exists(): 

292 out.print(f"[red]no such file:[/] {path}") 

293 raise typer.Exit(1) 

294 out.print(f"[dim]reading {path}[/]") 

295 

296 pushed: list[str] = [] 

297 skipped: list[str] = [] 

298 # Not interpolated: these values are going into permanent storage, where 

299 # a silently expanded ``${...}`` could not be recovered. 

300 for key, value in read_env_file(path, interpolate=False).items(): 

301 if key in SKIP_ON_IMPORT: 

302 skipped.append(key) 

303 continue 

304 if not value: 

305 continue 

306 if not dry_run: 

307 store.put(key, value) 

308 pushed.append(key) 

309 

310 verb = "would import" if dry_run else "imported" 

311 out.print(f"[green]{verb}[/] {len(pushed)} secret(s): {', '.join(pushed) or '—'}") 

312 if skipped: 

313 out.print(f"[dim]skipped AWS bootstrap keys: {', '.join(skipped)}[/]") 

314 

315 @app.command("export") 

316 def secrets_export(ctx: typer.Context) -> None: 

317 """Print every visible secret as KEY=VALUE lines, resolved along the path.""" 

318 store = store_from(ctx) 

319 for name in sorted(store.names()): 

320 emit(out, f"{name}={store.get(name, required=True)}") 

321 

322 return app
