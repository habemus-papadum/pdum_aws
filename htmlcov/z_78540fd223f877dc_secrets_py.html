1"""Secrets in AWS SSM Parameter Store, resolved along a search path. 

2 

3A :class:`SecretStore` is built from a **search path** of SSM prefixes, most 

4specific first — ``"/myapp/:/org/"`` — and maps a name like ``STRIPE_KEY`` 

5onto the parameter ``<prefix>STRIPE_KEY`` in each layer. Reads resolve in this 

6order: 

7 

81. a real **environment variable** of that name, so a shell export, a ``.env`` 

9 loaded by the caller, or a test fixture can override without touching AWS; 

102. each **SSM prefix** in path order (decrypted), first hit wins; 

113. the supplied ``default``, or a :class:`KeyError` when ``required=True``. 

12 

13Layered prefixes work like layered config files (project, then org, then 

14global): a project stores only what is truly its own and inherits the rest, 

15so a secret shared by many projects lives in exactly one place. **Writes and 

16deletes always target the first prefix** — the layer the store belongs to — 

17never a fallback layer; mutating a shared layer requires constructing a store 

18whose path starts there. 

19 

20Why Parameter Store rather than Secrets Manager: standard-tier parameters are 

21free where Secrets Manager is roughly $0.40 per secret per month, and both give 

22KMS encryption, IAM gating and CloudTrail audit. This module is the only thing 

23that knows which backend is in use, so switching later is a one-file change. 

24 

25The path is **required** and has no default — a shared default would let two 

26unrelated projects collide in the same namespace, and would make 

27:meth:`SecretStore.names` return parameters the caller does not own. 

28 

29```python 

30from pdum.aws.secrets import SecretStore 

31 

32store = SecretStore("/myapp/:/org/") 

33store.put("STRIPE_KEY", "sk_live_...") # written to /myapp/STRIPE_KEY 

34store.get("GOOGLE_CLIENT_ID") # found at /org/GOOGLE_CLIENT_ID 

35``` 

36 

37This module does not read ``.env`` files, and nothing here touches the 

38filesystem as a side effect of being imported. Env-first resolution looks at 

39``os.environ`` only. When you want the ``.env`` layer too, call 

40:func:`pdum.aws.env.load_env` explicitly — it applies the files and then this 

41store, in that order, and returns a report of what each contributed. 

42"""

54def _normalise_prefix(prefix: str) -> str: 

55 """Return *prefix* as a leading- and trailing-slashed path. 

56 

57 Accepts ``myapp``, ``/myapp`` or ``/myapp/`` and returns ``/myapp/`` for all 

58 three, so callers need not remember the convention. 

59 

60 Parameters 

61 ---------- 

62 prefix : str 

63 The desired SSM path prefix. 

64 

65 Returns 

66 ------- 

67 str 

68 Normalised prefix, e.g. ``"/myapp/"``. 

69 

70 Raises 

71 ------ 

72 ValueError 

73 If *prefix* is empty, blank, or ``"/"``. A root prefix would make 

74 :meth:`SecretStore.names` enumerate every parameter in the account, 

75 including ones belonging to other applications. 

76 """ 

77 trimmed = prefix.strip().strip("/") 

78 if not trimmed: 

79 raise ValueError( 

80 "SecretStore requires a non-root prefix such as '/myapp/'; " 

81 f"got {prefix!r}. A root prefix would span the whole account's Parameter Store." 

82 ) 

83 return f"/{trimmed}/"

86def _normalise_path(path: str | Sequence[str]) -> tuple[str, ...]: 

87 """Return *path* as a tuple of normalised prefixes, most specific first. 

88 

89 Accepts a colon-separated string (``"/myapp/:/org/"``) or a sequence of 

90 prefixes; each element is normalised by :func:`_normalise_prefix`. 

91 Duplicates are dropped, keeping the first occurrence. 

92 

93 Raises 

94 ------ 

95 ValueError 

96 If the path is empty, or any element is empty or root. 

97 """ 

98 parts = [p for p in path.split(":") if p.strip()] if isinstance(path, str) else list(path) 

99 if not parts: 

100 raise ValueError( 

101 "SecretStore requires a search path of at least one prefix, " 

102 f"such as '/myapp/' or '/myapp/:/org/'; got {path!r}." 

103 ) 

104 out: list[str] = [] 

105 for part in parts: 

106 prefix = _normalise_prefix(part) 

107 if prefix not in out: 

108 out.append(prefix) 

109 return tuple(out)

112class SecretStore: 

113 """A namespaced view of SSM Parameter Store, layered along a search path. 

114 

115 Parameters 

116 ---------- 

117 path : str or sequence of str 

118 SSM search path, most specific prefix first: a colon-separated string 

119 (``"/myapp/:/org/"``) or a sequence (``["/myapp/", "/org/"]``). 

120 Required; see the module docstring for why there is no default. 

121 region : str, optional 

122 Region for the SSM client. When ``None``, ``boto3`` resolves it from the 

123 environment or active profile. 

124 

125 Attributes 

126 ---------- 

127 prefixes : tuple of str 

128 The normalised search path, each element leading- and trailing-slashed. 

129 """ 

130 

131 def __init__(self, path: str | Sequence[str], *, region: str | None = None) -> None: 

132 self.prefixes = _normalise_path(path) 

133 self._region = region 

134 self._cache: dict[str, str | None] = {} 

135 self._ssm: Any = None 

136 

137 def __repr__(self) -> str: 

138 return f"{type(self).__name__}(path={':'.join(self.prefixes)!r})" 

139 

140 @property 

141 def prefix(self) -> str: 

142 """The first prefix on the path — the layer writes and deletes target.""" 

143 return self.prefixes[0]

145 @property 

146 def ssm(self) -> Any: 

147 """The SSM client, created lazily on first use. 

148 

149 Deferred so that constructing a store, or resolving a secret that is 

150 already present in the environment, never requires credentials. 

151 """ 

152 if self._ssm is None: 

153 self._ssm = client("ssm", self._region) 

154 return self._ssm

156 def path(self, name: str) -> str: 

157 """Return the full SSM parameter path for *name* in the write layer. 

158 

159 Parameters 

160 ---------- 

161 name : str 

162 Secret name without a prefix. 

163 

164 Returns 

165 ------- 

166 str 

167 The full path, e.g. ``"/myapp/STRIPE_KEY"``. 

168 """ 

169 return self.prefix + name

171 def _read_layer(self, prefix: str, name: str) -> str | None: 

172 """Fetch ``<prefix><name>`` from SSM, or ``None`` when absent.""" 

173 from botocore.exceptions import ClientError 

174 

175 try: 

176 response = self.ssm.get_parameter(Name=prefix + name, WithDecryption=True) 

177 except ClientError as exc: 

178 if exc.response["Error"]["Code"] == "ParameterNotFound": 

179 return None 

180 raise 

181 return response["Parameter"]["Value"]

183 def get(self, name: str, *, default: str | None = None, required: bool = False) -> str | None: 

184 """Resolve a secret from the environment, then each prefix, then *default*. 

185 

186 Successful and missing lookups are both cached for the life of the 

187 store, so repeated reads cost one walk of the path at most. 

188 

189 Parameters 

190 ---------- 

191 name : str 

192 Secret name without a prefix. 

193 default : str, optional 

194 Returned when the secret exists nowhere and ``required`` is false. 

195 required : bool, default False 

196 Raise instead of returning *default* when the secret is missing. 

197 

198 Returns 

199 ------- 

200 str or None 

201 The secret value, or *default* when absent. 

202 

203 Raises 

204 ------ 

205 KeyError 

206 If the secret is missing and ``required`` is true. 

207 """ 

208 env_value = os.environ.get(name) 

209 if env_value: 

210 return env_value 

211 

212 if name not in self._cache: 

213 value = None 

214 for prefix in self.prefixes: 

215 value = self._read_layer(prefix, name) 

216 if value is not None: 

217 break 

218 self._cache[name] = value 

219 

220 value = self._cache[name] 

221 if value is None: 

222 if required: 

223 searched = ", ".join(p + name for p in self.prefixes) 

224 raise KeyError(f"secret {name!r} not found in the environment or SSM at {searched}") 

225 return default 

226 return value

228 def get_json(self, name: str, *, default: Any = None, required: bool = False) -> Any: 

229 """Resolve a secret and parse it as JSON. 

230 

231 Parameters 

232 ---------- 

233 name : str 

234 Secret name without a prefix. 

235 default : Any, optional 

236 Returned when the secret is absent and ``required`` is false. 

237 required : bool, default False 

238 Raise instead of returning *default* when the secret is missing. 

239 

240 Returns 

241 ------- 

242 Any 

243 The parsed JSON value, or *default* when absent. 

244 

245 Raises 

246 ------ 

247 KeyError 

248 If the secret is missing and ``required`` is true. 

249 json.JSONDecodeError 

250 If the stored value is not valid JSON. 

251 """ 

252 raw = self.get(name, default=None, required=required) 

253 return json.loads(raw) if raw else default

255 def put(self, name: str, value: str, *, secure: bool = True) -> None: 

256 """Create or overwrite a secret in the write layer, updating the cache. 

257 

258 Parameters 

259 ---------- 

260 name : str 

261 Secret name without a prefix. 

262 value : str 

263 Value to store. 

264 secure : bool, default True 

265 Store as a KMS-encrypted ``SecureString``. Pass ``False`` for 

266 non-sensitive configuration that benefits from the same namespace. 

267 """ 

268 self.ssm.put_parameter( 

269 Name=self.path(name), 

270 Value=value, 

271 Type="SecureString" if secure else "String", 

272 Overwrite=True, 

273 ) 

274 self._cache[name] = value

276 def delete(self, name: str) -> None: 

277 """Delete a secret from the write layer and drop it from the cache. 

278 

279 Only the first prefix is ever deleted from. When the name exists solely 

280 in a fallback layer, this raises rather than reaching down: deleting 

281 from a shared layer must be asked for explicitly, with a store whose 

282 path starts there. 

283 

284 Parameters 

285 ---------- 

286 name : str 

287 Secret name without a prefix. 

288 

289 Raises 

290 ------ 

291 KeyError 

292 If the secret is not in the write layer. The message names the 

293 fallback layer holding it, when there is one. 

294 """ 

295 from botocore.exceptions import ClientError 

296 

297 try: 

298 self.ssm.delete_parameter(Name=self.path(name)) 

299 except ClientError as exc: 

300 if exc.response["Error"]["Code"] != "ParameterNotFound": 

301 raise 

302 for fallback in self.prefixes[1:]: 

303 if self._read_layer(fallback, name) is not None: 

304 raise KeyError( 

305 f"secret {name!r} is not in the write layer {self.prefix}; it lives at " 

306 f"{fallback + name} — use a store with path {fallback!r} to delete it there" 

307 ) from exc 

308 raise KeyError(f"secret {name!r} not found under {':'.join(self.prefixes)}") from exc 

309 self._cache.pop(name, None)

314 Names present in several layers appear once. 

315 

316 Returns

318 list of str

323 def read_all(self) -> dict[str, str]: 

324 """Read every secret on the search path in one pass, first layer winning. 

325 

326 Costs one paginated call per layer rather than one call per name, which 

327 is what makes materialising a whole store — into a process environment, 

328 say — cheap enough to do on every invocation. Unlike :meth:`describe` 

329 this decrypts, so it does pull values over the wire. 

330 

331 The environment is deliberately **not** consulted, unlike :meth:`get`: 

332 this is the store's own view, leaving the caller to decide how it should 

333 interact with variables that are already set. 

334 

335 Values are cached as if each name had been fetched individually, so a 

336 later :meth:`get` for any of them costs nothing. 

337 

338 Returns 

339 ------- 

340 dict of str to str 

341 Secret name to value, shadowed layers excluded. 

342 """ 

343 paginator = self.ssm.get_paginator("get_parameters_by_path") 

344 out: dict[str, str] = {} 

345 for prefix in self.prefixes: 

346 for page in paginator.paginate(Path=prefix, Recursive=True, WithDecryption=True): 

347 for p in page["Parameters"]: 

348 out.setdefault(p["Name"][len(prefix) :], p["Value"]) 

349 self._cache.update(out) 

350 return out

352 def describe(self) -> list[dict[str, Any]]: 

353 """List metadata for each secret on the search path. 

354 

355 Deliberately does not decrypt: this reads names, types and timestamps 

356 only, so rendering a listing never pulls secret values over the wire. 

357 

358 Returns 

359 ------- 

360 list of dict 

361 One dict per parameter per layer, with ``name``, ``type``, 

362 ``version``, ``modified``, ``origin`` (the prefix holding it) and 

363 ``shadowed`` (true when an earlier layer also has the name, so 

364 resolution never reaches this row). 

365 """ 

366 paginator = self.ssm.get_paginator("get_parameters_by_path") 

367 out: list[dict[str, Any]] = [] 

368 seen: set[str] = set() 

369 for prefix in self.prefixes: 

370 for page in paginator.paginate(Path=prefix, Recursive=True): 

371 for p in page["Parameters"]: 

372 name = p["Name"][len(prefix) :] 

373 out.append( 

374 { 

375 "name": name, 

376 "type": p["Type"], 

377 "version": p["Version"], 

378 "modified": p["LastModifiedDate"], 

379 "origin": prefix, 

380 "shadowed": name in seen, 

381 } 

382 ) 

383 seen.update(row["name"] for row in out) 

384 return out
