1"""``pdx`` — run a command with a project's secrets in its environment. 

2 

3```console 

4$ pdx npm run dev 

5$ pdx python -m my_service --port 8080 

6$ pdx -- ls -la 

7``` 

8 

9**``pdx`` has no options of its own.** Everything after the name is the command 

10to run, so no flag of yours can ever collide with one of its own, and ``--`` is 

11available but never required. The only exception is a leading ``--help``, which 

12click reserves; ``pdx python --help`` still reaches Python, because parsing 

13stops at the first bare word. 

14 

15Configuration therefore comes entirely from the environment: the search path is 

16``PDUM_SSM_PATH`` if set, otherwise a ``PDUM_SSM_PATH=`` line in the nearest 

17``.env`` — searched upward from the working directory, so this works from 

18anywhere inside a project. Every secret on that path is copied into the 

19environment, and then the process is **replaced** by the command, a real 

20``execvp``: same PID, so signals, job control, exit status and the terminal all 

21belong to the program you asked for, with nothing left in the middle to forward 

22them. 

23 

24Variables already set are left alone. Your shell is more specific than the 

25store, so ``STRIPE_KEY=sk_test_... pdx ./run-tests`` overrides for that one run, 

26and a CI job's injected variables are never undone. ``pdx-doctor`` shows what 

27that adds up to without running anything. 

28 

29Only the ``PDUM_SSM_PATH`` line of a ``.env`` is read. The file says *where the 

30secrets live*; it is not a second source of secrets, since then a value's origin 

31would be ambiguous. Use ``secrets import`` to move a ``.env`` into the store. 

32 

33Both commands are attachable, so an application can offer the same thing under 

34its own name and defaults: 

35 

36```python 

37from pdum.aws.cli import add_pdx, add_pdx_doctor 

38 

39add_pdx(app, default_path="/acme/:/org/", envvar="ACME_SSM_PATH") 

40add_pdx_doctor(app, default_path="/acme/:/org/", envvar="ACME_SSM_PATH") 

41``` 

42"""

69#: Names a shell can actually use. A store may hold nested parameters, whose 

70#: names carry a slash (``db/PASSWORD``); exporting one would make a variable no 

71#: program could name, so it is reported and skipped instead. 

72_ENV_NAME = re.compile(r"[A-Za-z_][A-Za-z0-9_]*\Z") 

73 

74#: Click settings that keep this command's parser off the wrapped command line. 

75#: ``allow_interspersed_args`` stops option parsing at the first bare word, and 

76#: ``ignore_unknown_options`` keeps ``pdx python -c ...`` from being read as an 

77#: unknown ``-c`` flag. Together they make ``--`` optional rather than required. 

78_PASSTHROUGH = {"ignore_unknown_options": True, "allow_interspersed_args": False} 

79 

80#: The codes a shell uses for these two conditions, worth matching so that 

81#: ``pdx some-typo`` is diagnosable exactly as a bare ``some-typo`` would be. 

82EXIT_NOT_FOUND = 127 

83EXIT_NOT_EXECUTABLE = 126 

84 

85#: A search path supplied by the host: literal, deferred, or absent. 

86PathDefault = str | Callable[[], str | None] | None

108def resolve_search_path( 

109 *, 

110 envvar: str | None, 

111 env_file: str, 

112 default_path: PathDefault, 

113 environ: Mapping[str, str], 

114) -> tuple[str | None, str]: 

115 """Work out which SSM search path to use, and say where it came from. 

116 

117 The environment wins over the file, so a variable exported for one command 

118 overrides the project's own setting the way it would for any other tool. 

119 

120 Parameters 

121 ---------- 

122 envvar : str, optional 

123 Variable holding the path, and the key looked for in the file. ``None`` 

124 disables both. 

125 env_file : str 

126 File to search for upward from the working directory. 

127 default_path : str or callable, optional 

128 Fallback when neither supplies one; a callable is resolved here. 

129 environ : mapping 

130 Environment to consult, normally ``os.environ``. 

131 

132 Returns 

133 ------- 

134 tuple of (str or None, str) 

135 The path, and a short description of its source for reporting. 

136 """ 

137 if envvar and environ.get(envvar): 

138 return environ[envvar], f"${envvar}" 

139 if envvar: 

140 found = find_env_file(env_file) 

141 if found is not None and (value := read_env_file(found).get(envvar)): 

142 return value, str(found) 

143 resolved = default_path() if callable(default_path) else default_path 

144 return (resolved, "built-in default") if resolved else (None, "")

147def load_environment(store: SecretStore, environ: MutableMapping[str, str]) -> LoadReport: 

148 """Copy *store* into *environ*, without disturbing what is already there. 

149 

150 Parameters 

151 ---------- 

152 store : SecretStore 

153 Store to read, in one bulk pass. 

154 environ : mutable mapping 

155 Environment to update, normally ``os.environ``. 

156 

157 Returns 

158 ------- 

159 LoadReport 

160 What was applied, kept and skipped. 

161 """ 

162 applied: list[str] = [] 

163 already_set: list[str] = [] 

164 skipped: list[str] = [] 

165 for name, value in sorted(store.read_all().items()): 

166 if not _ENV_NAME.match(name): 

167 skipped.append(name) 

168 elif environ.get(name): 

169 # Truthiness, not membership, to agree with SecretStore.get, where 

170 # an empty variable reads as unset. 

171 already_set.append(name) 

172 else: 

173 environ[name] = value 

174 applied.append(name) 

175 return LoadReport(applied, already_set, skipped)

178def _no_path_message(envvar: str | None, env_file: str) -> str: 

179 """Explain every way the search path could have been supplied.""" 

180 ways = [f"set {envvar}", f"add {envvar}=/myapp/:/org/ to {env_file}"] if envvar else [] 

181 return f"[red]no SSM search path configured:[/] {', or '.join(ways) or 'this command was built without one'}"

184def _exec(argv: list[str], console: Console) -> NoReturn: 

185 """Replace this process with *argv*. 

186 

187 Buffers are flushed first: ``exec`` runs no ``atexit`` handlers and flushes 

188 nothing still sitting in userspace, so anything already written would 

189 otherwise be lost. 

190 

191 Raises 

192 ------ 

193 SystemExit 

194 :data:`EXIT_NOT_FOUND` when the command is not on ``PATH``, 

195 :data:`EXIT_NOT_EXECUTABLE` when it is there but cannot be run. 

196 """ 

197 sys.stdout.flush() 

198 sys.stderr.flush() 

199 if hasattr(console.file, "flush"): 

200 console.file.flush() 

201 try: 

202 os.execvp(argv[0], argv) 

203 except FileNotFoundError: 

204 console.print(f"[red]{argv[0]}: command not found[/]") 

205 raise SystemExit(EXIT_NOT_FOUND) from None 

206 except OSError as exc: 

207 console.print(f"[red]{argv[0]}: {exc.strerror or exc}[/]") 

208 raise SystemExit(EXIT_NOT_EXECUTABLE) from None

211def add_pdx( 

212 app: typer.Typer, 

213 *, 

214 console: Console | None = None, 

215 default_path: PathDefault = None, 

216 envvar: str | None = "PDUM_SSM_PATH", 

217 env_file: str = ".env", 

218 store_factory: Callable[[str], SecretStore] = SecretStore, 

219 name: str = "pdx", 

220) -> None: 

221 """Register a ``pdx`` command on *app*. 

222 

223 Parameters 

224 ---------- 

225 app : typer.Typer 

226 Application to register the command on. 

227 console : rich.console.Console, optional 

228 Console for this command's own diagnostics. Defaults to 

229 :data:`~pdum.aws.cli.output.DEFAULT_STDERR_CONSOLE`; whatever you pass 

230 should write to stderr, since stdout belongs to the wrapped program. 

231 default_path : str or callable, optional 

232 Search path used when neither the environment nor the ``.env`` supplies 

233 one. A callable is resolved at invocation. 

234 envvar : str, optional 

235 Environment variable holding the search path, and the key looked for in 

236 the ``.env``. ``None`` disables both, leaving only *default_path*. 

237 env_file : str, default ".env" 

238 File searched for upward from the working directory. A name with a 

239 directory component is used as given. 

240 store_factory : callable, default :class:`~pdum.aws.secrets.SecretStore` 

241 Called with the resolved path to build the store. 

242 name : str, default "pdx" 

243 Name to register the command under. 

244 """ 

245 out = console or DEFAULT_STDERR_CONSOLE 

246 

247 @app.command(name, context_settings=_PASSTHROUGH) 

248 def pdx( 

249 command: list[str] = typer.Argument(None, metavar="COMMAND [ARGS]...", help="program to run, with arguments"), 

250 ) -> None: 

251 """Run a command with this project's secrets in its environment. 

252 

253 Takes no options of its own: everything here is the command to run. 

254 Variables already set are left alone. 

255 """ 

256 argv = list(command or []) 

257 if not argv: 

258 out.print("[red]nothing to run:[/] pdx takes the command to exec, e.g. [bold]pdx npm start[/]") 

259 raise typer.Exit(2) 

260 

261 path, _ = resolve_search_path(envvar=envvar, env_file=env_file, default_path=default_path, environ=os.environ) 

262 if not path: 

263 out.print(_no_path_message(envvar, env_file)) 

264 raise typer.Exit(2) 

265 try: 

266 store = store_factory(path) 

267 except ValueError as exc: 

268 out.print(f"[red]{exc}[/]") 

269 raise typer.Exit(2) from exc 

270 

271 load_environment(store, os.environ) 

272 _exec(argv, out)

275def add_pdx_doctor( 

276 app: typer.Typer, 

277 *, 

278 console: Console | None = None, 

279 default_path: PathDefault = None, 

280 envvar: str | None = "PDUM_SSM_PATH", 

281 env_file: str = ".env", 

282 store_factory: Callable[[str], SecretStore] = SecretStore, 

283 name: str = "pdx-doctor", 

284) -> None: 

285 """Register a command explaining what :func:`add_pdx` would do. 

286 

287 Answers the questions ``secrets list`` cannot, because they are about this 

288 process rather than about the store: which search path resolves and *from 

289 where*, and which names the environment already holds — the reason a program 

290 run under ``pdx`` can see a value that is not the one in SSM. 

291 

292 Deliberately reports names and origins only, never values: it works from 

293 :meth:`~pdum.aws.secrets.SecretStore.describe`, which does not decrypt, so 

294 running it pulls no secret over the wire. Use ``secrets get`` when the value 

295 is what you actually want. 

296 

297 Parameters are as for :func:`add_pdx`, except that *console* defaults to 

298 stdout — here the report is the output. 

299 """ 

300 out = console or DEFAULT_CONSOLE 

301 

302 @app.command(name) 

303 def pdx_doctor() -> None: 

304 """Explain what pdx would put in the environment, without running anything.""" 

305 path, source = resolve_search_path( 

306 envvar=envvar, env_file=env_file, default_path=default_path, environ=os.environ 

307 ) 

308 if not path: 

309 out.print(_no_path_message(envvar, env_file)) 

310 raise typer.Exit(2) 

311 out.print(f"[bold]search path[/] {path} [dim](from {source})[/]") 

312 try: 

313 store = store_factory(path) 

314 except ValueError as exc: 

315 out.print(f"[red]{exc}[/]") 

316 raise typer.Exit(2) from exc 

317 

318 rows = sorted((r for r in store.describe() if not r["shadowed"]), key=lambda r: r["name"]) 

319 if not rows: 

320 out.print(f"[yellow]no secrets under {path}[/]") 

321 return 

322 

323 table = Table(box=None, pad_edge=False) 

324 table.add_column("NAME", style="bold") 

325 table.add_column("LAYER") 

326 table.add_column("PDX WOULD") 

327 counts = {"set it": 0, "keep the environment's": 0, "skip it": 0} 

328 for row in rows: 

329 name_ = row["name"] 

330 if not _ENV_NAME.match(name_): 

331 verdict, style = "skip it", "yellow" 

332 elif os.environ.get(name_): 

333 verdict, style = "keep the environment's", "cyan" 

334 else: 

335 verdict, style = "set it", "green" 

336 counts[verdict] += 1 

337 table.add_row(name_, row["origin"], f"[{style}]{verdict}[/]") 

338 out.print(table) 

339 out.print(" [dim]" + ", ".join(f"{n} to {verdict}" for verdict, n in counts.items() if n) + "[/]")
